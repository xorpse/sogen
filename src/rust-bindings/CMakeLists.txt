# Collects everything a final link of `item` needs, in reverse post-order so that every library precedes its
# dependencies, as single-pass linkers require.
function(sogen_rust_collect_link_items item)
  if(item MATCHES "^::@" OR item MATCHES "^\\$<INSTALL_INTERFACE:")
    return()
  endif()

  while(item MATCHES "^\\$<(LINK_ONLY|BUILD_INTERFACE|TARGET_NAME):(.+)>$")
    set(item "${CMAKE_MATCH_2}")
  endwhile()

  if(item MATCHES "^\\$<\\$<BOOL:([^>]*)>:(.+)>$")
    set(value "${CMAKE_MATCH_2}")
    if(CMAKE_MATCH_1)
      sogen_rust_collect_link_items("${value}")
    endif()
    return()
  endif()

  if(item MATCHES "^\\$<LINK_LIBRARY:([A-Za-z_]+),(.+)>$")
    set(feature "${CMAKE_MATCH_1}")
    string(REPLACE "," ";" libraries "${CMAKE_MATCH_2}")
    foreach(library IN LISTS libraries)
      if(feature MATCHES "FRAMEWORK$")
        set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "item=-framework ${library}")
      else()
        sogen_rust_collect_link_items("${library}")
      endif()
    endforeach()
    return()
  endif()

  get_property(visited GLOBAL PROPERTY SOGEN_RUST_LINK_VISITED)
  if("${item}" IN_LIST visited)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_VISITED "${item}")

  if(NOT TARGET "${item}")
    if(item MATCHES "::" AND NOT item MATCHES "^\\$<")
      message(FATAL_ERROR "${item} is not visible from ${CMAKE_CURRENT_SOURCE_DIR}; import it here so the Rust link manifest can resolve it")
    endif()

    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "item=${item}")
    return()
  endif()

  get_target_property(aliased "${item}" ALIASED_TARGET)
  if(aliased)
    sogen_rust_collect_link_items("${aliased}")
    return()
  endif()

  get_target_property(type "${item}" TYPE)
  get_target_property(imported "${item}" IMPORTED)

  set(dependencies)
  get_target_property(interface_libraries "${item}" INTERFACE_LINK_LIBRARIES)
  if(interface_libraries)
    list(APPEND dependencies ${interface_libraries})
  endif()

  if(type STREQUAL "STATIC_LIBRARY" AND NOT imported)
    get_target_property(private_libraries "${item}" LINK_LIBRARIES)
    if(private_libraries)
      list(APPEND dependencies ${private_libraries})
    endif()
  endif()

  foreach(dependency IN LISTS dependencies)
    sogen_rust_collect_link_items("${dependency}")
  endforeach()

  if(item STREQUAL "sogen-rust-shim")
    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "bundle=$<TARGET_LINKER_FILE:${item}>")
  elseif(type STREQUAL "STATIC_LIBRARY")
    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "static=$<TARGET_LINKER_FILE:${item}>")
  elseif(type STREQUAL "SHARED_LIBRARY")
    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "shared=$<TARGET_LINKER_FILE:${item}>")
    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_RUNTIME_DIRS "$<TARGET_FILE_DIR:${item}>")
  elseif(type STREQUAL "UNKNOWN_LIBRARY")
    set_property(GLOBAL APPEND PROPERTY SOGEN_RUST_LINK_ITEMS "item=$<TARGET_FILE:${item}>")
  endif()
endfunction()

set_property(GLOBAL PROPERTY SOGEN_RUST_LINK_VISITED)
set_property(GLOBAL PROPERTY SOGEN_RUST_LINK_ITEMS)
set_property(GLOBAL PROPERTY SOGEN_RUST_RUNTIME_DIRS)
sogen_rust_collect_link_items(sogen-rust-shim)
get_property(LINK_ITEMS GLOBAL PROPERTY SOGEN_RUST_LINK_ITEMS)
get_property(RUNTIME_DIRS GLOBAL PROPERTY SOGEN_RUST_RUNTIME_DIRS)
list(REVERSE LINK_ITEMS)
list(JOIN LINK_ITEMS "\n" LINK_MANIFEST)
list(REMOVE_DUPLICATES RUNTIME_DIRS)
list(JOIN RUNTIME_DIRS "\n" RUNTIME_MANIFEST)

file(GENERATE
  OUTPUT "${CMAKE_BINARY_DIR}/sogen-rust-link-$<CONFIG>.txt"
  CONTENT "${LINK_MANIFEST}\n"
)
file(GENERATE
  OUTPUT "${CMAKE_BINARY_DIR}/sogen-rust-runtime-$<CONFIG>.txt"
  CONTENT "${RUNTIME_MANIFEST}\n"
)
